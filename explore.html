viewof pickedGenres = Inputs.checkbox(
  ["Horror", "Comedy", "Action", "Adventure", "Drama"],
  {label: "Genres", value: ["Horror", "Comedy"]}
)
viewof startYear = Inputs.range([1970, 2019], {label: "From year", step: 1, value: 1990})
viewof search = Inputs.text({label: "Search a title", placeholder: "Try Scream or Bridesmaids"})
data = transpose(movies)
  .filter(d => pickedGenres.includes(d.genre))
  .filter(d => d.year >= startYear)
  .filter(d => d.movie.toLowerCase().includes(search.toLowerCase()))
Plot.plot({
  width: 800,
  height: 520,
  marginLeft: 70,
  marginBottom: 50,
  style: {background: "transparent", color: "#f1f1f1", fontSize: "13px"},
  x: {type: "log", label: "Production budget →", labelAnchor: "center", labelOffset: 42, ticks: 5, tickFormat: "$~s"},
  y: {type: "log", label: "↑ Worldwide box office", ticks: 6, tickFormat: "$~s"},
  color: {
    domain: ["Horror", "Comedy", "Action", "Adventure", "Drama"],

  },

Inputs.table(
  data.slice().sort((a, b) => b.return_ratio - a.return_ratio),
  {
    columns: ["movie", "year", "genre", "production_budget", "worldwide_gross", "return_ratio"],
    header: {movie: "Movie", year: "Year", genre: "Genre", production_budget: "Budget",
             worldwide_gross: "Worldwide gross", return_ratio: "Times budget"},
    format: {
      year: d => d,
      production_budget: d => "$" + d3.format(",")(d),
      worldwide_gross: d => "$" + d3.format(",")(d),
      return_ratio: d => d.toFixed(1) + "x"
    },
    rows: 12
  }
)
